\begin{frame}{2. What frontier is expanding?}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \pill{Production frontier}\par\vspace{0.35cm}
      \large The set of languages in which a developer can \alert{ship code now}.
      \vspace{0.55cm}
      \begin{itemize}
        \item familiar set $\mathcal K_i$
        \item unfamiliar set $\mathcal U_i$
        \item active count $N^g_{it}=\sum_k Z^g_{ik,t}$
      \end{itemize}
    \column{0.47\textwidth}
      \pill{Not necessarily learning}\par\vspace{0.35cm}
      \large Delegation may activate a language without raising the developer's language-specific skill $s$.
      \vspace{0.55cm}
      \begin{itemize}
        \item developer specifies and verifies
        \item agent executes a share $\lambda$
        \item learning is a later, ambiguous signal channel
      \end{itemize}
  \end{columns}
  \source{Quispe and Xu, Section 3 and Appendix learning extension.}
  \note{Timing: 1:10. Keep production capability separate from human capital. This distinction prevents an apparent contradiction with deskilling models. [Sources] Quispe--Xu Sections 3.1 and 3.4; Appendix learning extension.}
\end{frame}

\begin{frame}{4. Entry is a threshold comparison}
  \small
  \begin{align*}
    T^S &= b-s\mu+\frac{\rho s^2}{2\pi},
    &T^C&=T^S-(\gamma s-r_C),\\
    T^1 &= \min\{T^S,T^C\}=T^S-\max\{0,\gamma s-r_C\},\\
    T^D &= b-(1-\lambda)s\mu-\lambda a z(A)+\kappa(a,s)+r_D
      +\frac{\rho}{2}\left[\frac{(1-\lambda)^2s^2}{\pi}+\sigma_D^2\right],\\
    T^2 &= \min\{T^1,T^D\}\le T^1.
  \end{align*}
  \vspace{0.25cm}
  \resultbox{A language is active iff $\omega\ge T^g$. Menu nesting, $\{S,C\}\subset\{S,C,D\}$, mechanically proves $Z^2\ge Z^1$ and $N^2\ge N^1$ path by path.}
  \source{Quispe and Xu, Equations (4)--(7), Proposition 1.}
  \note{Timing: 1:25. Derive by setting each certainty equivalent greater than or equal to zero. Distinguish the weak theorem from strict behavior change. [Sources] Quispe--Xu Section 3.2, Equations 4--7, Proposition 1.}
\end{frame}

\begin{frame}{9. From a first-use flow to a cumulative stock}
  \small
  For an initially unfamiliar language with first-use hazard $p^g_{ik}$,
  \[
    \Delta C_i(s)=\sum_{k\in\mathcal U_i}
      \left[(1-p^1_{ik})^{s+1}-(1-p^2_{ik})^{s+1}\right]\ge0
      \quad\text{if }p^2_{ik}\ge p^1_{ik}.
  \]
  \vspace{0.25cm}
  \begin{center}
  \begin{tikzpicture}[node distance=0.8cm]
    \node[draw=navy,rounded corners,fill=paper,minimum width=3cm,minimum height=1cm] (opp) {opportunity arrives};
    \node[draw=orange,rounded corners,fill=orange!8,minimum width=3cm,minimum height=1cm,right=of opp] (first) {first use (flow)};
    \node[draw=teal,rounded corners,fill=pale,minimum width=3cm,minimum height=1cm,right=of first] (stock) {ever used (stock)};
    \draw[-{Latex},line width=1.2pt,navy] (opp)--(first);
    \draw[-{Latex},line width=1.2pt,navy] (first)--(stock);
  \end{tikzpicture}
  \end{center}
  \vspace{0.25cm}
  \resultbox{Strict increase and strict discrete concavity do \emph{not} follow from $p^2\ge p^1$ alone. A sufficient nondegenerate benchmark is $p^1=0<p^2<1$; if $p^2=1$, the stock jumps once and then is flat.}
  \source{Quispe and Xu, Proposition 3 and Appendix proof.}
  \note{Timing: 1:25. Explain why new-language flows can spike and fade while cumulative breadth keeps rising. [Sources] Quispe--Xu Proposition 3 and Appendix proof.}
\end{frame}

\begin{frame}{12. Same AI, opposite conclusions: one modeling fork}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=0.8em]{block body}
        \textbf{Quispe--Xu}\par\vspace{0.3cm}
        \[\{S,C\}\longrightarrow\{S,C,D\}\]
        AI creates an optional delegation mode. General ability supports specification and verification; interaction can later generate information signals.
        \vspace{0.3cm}
        \textbf{Prediction:} lower language-entry thresholds, frontier expansion.
      \end{beamercolorbox}
    \column{0.48\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=0.8em]{block body}
        \textbf{Aouad--Lykouris--Zhong}\par\vspace{0.3cm}
        \[x=s+e+a\]
        AI assistance, skill, and effort are perfect substitutes. More $a$ crowds out effort; effort governs future skill transitions.
        \vspace{0.3cm}
        \textbf{Prediction:} deskilling and possible long-run productivity decline.
      \end{beamercolorbox}
  \end{columns}
  \vspace{0.35cm}
  \centering\alert{The core contrast is menu expansion versus input substitution. Information is secondary in Quispe--Xu.}
  \source{Quispe and Xu, Section 3 and learning appendix; Aouad, Lykouris, and Zhong, Sections 2--4.}
